const CITSPACE_CLUSTER_LAYOUT=[{x:58,y:46},{x:50,y:74},{x:82,y:72},{x:84,y:26},{x:42,y:44},{x:22,y:72},{x:18,y:24},{x:74,y:52}];
function pickCitespaceLayoutStyle(direction){if(direction==="数字法治"||direction==="知识产权法")return"vertical";if(direction==="刑事法治"||direction==="司法制度")return"horizontal";return"fishbone"}
function layoutCitespaceHorizontal(nodes,cx,cy){const ordered=nodes.slice().sort((a,b)=>a.avgYear-b.avgYear||a.minYear-b.minYear),n=ordered.length,span=Math.max(16,(n-1)*7.2),startX=cx-span/2;return ordered.map((node,index)=>({...node,tx:clamp(n<=1?cx:startX+(span*index)/Math.max(1,n-1),8,92),ty:clamp(cy,12,86),role:"spine",spineIndex:index,branchSide:null}))}
function layoutCitespaceVertical(nodes,cx,cy){const ordered=nodes.slice().sort((a,b)=>a.avgYear-b.avgYear||a.minYear-b.minYear),n=ordered.length,span=Math.max(14,(n-1)*5.8),startY=cy-span/2;return ordered.map((node,index)=>({...node,tx:clamp(cx,10,90),ty:clamp(n<=1?cy:startY+(span*index)/Math.max(1,n-1),12,86),role:"spine",spineIndex:index,branchSide:null}))}
function layoutCitespaceFishbone(nodes,cx,cy){const ordered=nodes.slice().sort((a,b)=>a.avgYear-b.avgYear||a.minYear-b.minYear||b.count-a.count),n=ordered.length,spineY=cy;if(n<=3)return layoutCitespaceHorizontal(nodes,cx,cy);const spineSlots=[0];if(n>1)spineSlots.push(n-1);for(let i=1;i<n-1;i+=1){if(i%2===0||spineSlots.length<3)spineSlots.push(i)}const uniqueSpine=[...new Set(spineSlots)].sort((a,b)=>a-b),spineSpan=Math.max(16,(uniqueSpine.length-1)*7.2),startX=cx-spineSpan/2,placed=new Map(),spineIndexByOrig=new Map();uniqueSpine.forEach((origIndex,si)=>{const node=ordered[origIndex];spineIndexByOrig.set(origIndex,si);placed.set(node,{tx:clamp(uniqueSpine.length<=1?cx:startX+(spineSpan*si)/Math.max(1,uniqueSpine.length-1),8,92),ty:clamp(spineY,12,86),role:"spine",spineIndex:si,branchSide:null})});ordered.forEach((node,index)=>{if(placed.has(node))return;let nearest=uniqueSpine[0],best=Math.abs(index-nearest);uniqueSpine.forEach(slot=>{const gap=Math.abs(index-slot);if(gap<best){best=gap;nearest=slot}});const parent=placed.get(ordered[nearest]),si=spineIndexByOrig.get(nearest),above=index%2===1;placed.set(node,{tx:parent.tx,ty:clamp(spineY+(above?-6.4:6.4),12,86),role:"branch",parentSpineIndex:si,branchSide:above?"above":"below"})});return ordered.map(node=>Object.assign({},node,placed.get(node)))}
function layoutCitespaceCluster(nodes,cx,cy,style){if(style==="vertical")return layoutCitespaceVertical(nodes,cx,cy);if(style==="horizontal")return layoutCitespaceHorizontal(nodes,cx,cy);return layoutCitespaceFishbone(nodes,cx,cy)}
function buildCitespaceTrendLinks(nodes){const links=[],spine=nodes.filter(node=>node.role==="spine").sort((a,b)=>a.spineIndex-b.spineIndex),branches=nodes.filter(node=>node.role==="branch");for(let i=0;i<spine.length-1;i+=1)links.push({source:spine[i],target:spine[i+1],kind:"spine"});branches.forEach(branch=>{const parent=spine[branch.parentSpineIndex]||spine[0];if(parent&&parent!==branch)links.push({source:parent,target:branch,kind:"branch"})});if(!links.length&&nodes.length>1)for(let i=0;i<nodes.length-1;i+=1)links.push({source:nodes[i],target:nodes[i+1],kind:"spine"});return links}
function buildClusterHullPoints(nodes,centerX,centerY,titleY,title){let minX=centerX,maxX=centerX,minY=Math.min(centerY,titleY),maxY=Math.max(centerY,titleY);nodes.forEach(node=>{const halfW=Math.max(5.4,String(node.label).length*.72+3);const labelY=node.ty+(node.role==="branch"?(node.branchSide==="above"?-2.6:2.6):1.8);minX=Math.min(minX,node.tx-halfW);maxX=Math.max(maxX,node.tx+halfW);minY=Math.min(minY,node.ty-2.4,labelY-2);maxY=Math.max(maxY,node.ty+2.4,labelY+5.6)});const titleHalfW=Math.min(28,String(title).length*.7+5);minX=Math.min(minX,centerX-titleHalfW);maxX=Math.max(maxX,centerX+titleHalfW);minY=Math.min(minY,titleY-3.6);maxY=Math.max(maxY,titleY+3.4);const padX=3.4,padY=3.2;minX-=padX;maxX+=padX;minY-=padY;maxY+=padY;const cx=(minX+maxX)/2,cy=(minY+maxY)/2,rx=Math.max((maxX-minX)/2,7),ry=Math.max((maxY-minY)/2,6.5);return Array.from({length:16},(_,index)=>{const angle=(Math.PI*2*index)/16-Math.PI/2,wobble=1+((index%4)===0?.035:0);return{x:clamp(cx+Math.cos(angle)*rx*wobble,3,97),y:clamp(cy+Math.sin(angle)*ry*wobble,5,95)}})}
function clusterBoundsFromHull(blobPoints){const xs=blobPoints.map(point=>point.x),ys=blobPoints.map(point=>point.y);return{cx:(Math.min(...xs)+Math.max(...xs))/2,cy:(Math.min(...ys)+Math.max(...ys))/2,rx:(Math.max(...xs)-Math.min(...xs))/2,ry:(Math.max(...ys)-Math.min(...ys))/2}}
function shiftCitespaceCluster(cluster,dx,dy){cluster.centerX=clamp(cluster.centerX+dx,10,90);cluster.centerY=clamp(cluster.centerY+dy,10,86);cluster.titleY=clamp(cluster.titleY+dy,10,88);cluster.nodes.forEach(node=>{node.tx=clamp(node.tx+dx,5,95);node.ty=clamp(node.ty+dy,6,94)});cluster.blobPoints=cluster.blobPoints.map(point=>({x:clamp(point.x+dx,3,97),y:clamp(point.y+dy,5,95)}));Object.assign(cluster,clusterBoundsFromHull(cluster.blobPoints))}
function separateCitespaceClusters(clusters){clusters.forEach(cluster=>{cluster.blobPoints=buildClusterHullPoints(cluster.nodes,cluster.centerX,cluster.centerY,cluster.titleY,cluster.title);Object.assign(cluster,clusterBoundsFromHull(cluster.blobPoints))});for(let step=0;step<22;step+=1){for(let i=0;i<clusters.length;i+=1){for(let j=i+1;j<clusters.length;j+=1){const a=clusters[i],b=clusters[j],dx=b.cx-a.cx,dy=b.cy-a.cy,dist=Math.hypot(dx,dy)||.01,minDist=a.rx+b.rx+9;if(dist>=minDist)continue;const push=(minDist-dist)*.32,fx=dx/dist*push,fy=dy/dist*push;shiftCitespaceCluster(a,-fx,-fy);shiftCitespaceCluster(b,fx,fy)}}}return clusters}
