function timelineDirectedLinkMarkup(s,t,stroke,thin){const dx=t.x-s.x,dy=t.y-s.y,dist=Math.hypot(dx,dy)||1,ux=dx/dist,uy=dy/dist,trim=thin?8:10,fromX=s.x+ux*trim,fromY=s.y+uy*trim,toX=t.x-ux*trim,toY=t.y-uy*trim,px=-uy,py=ux,arrowSize=thin?6:7,leftX=toX-ux*arrowSize+px*arrowSize*.7,leftY=toY-uy*arrowSize+py*arrowSize*.7,rightX=toX-ux*arrowSize-px*arrowSize*.7,rightY=toY-uy*arrowSize-py*arrowSize*.7,w=thin?1.35:2.1;return`<line x1="${fromX}" y1="${fromY}" x2="${toX}" y2="${toY}" stroke="${stroke}" stroke-width="${w}" stroke-opacity=".68"></line><path d="M ${toX} ${toY} L ${leftX} ${leftY} L ${rightX} ${rightY} Z" fill="${stroke}" fill-opacity=".9"></path>`}
